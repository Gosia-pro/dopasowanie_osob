#Wizualne ulepszenia
st.set_page_config(page_title="Dopasowanie osób", layout="wide")

MODEL_NAME = 'welcome_survey_clustering_pipeline_v1'

DATA = 'welcome_survey_simple_v1.csv'

CLUSTER_NAMES_AND_DESCRIPTIONS = 'welcome_survey_cluster_names_and_descriptions_v1.json'


@st.cache_data
def get_model():
    return load_model(MODEL_NAME)

@st.cache_data
def get_cluster_names_and_descriptions():
    with open(CLUSTER_NAMES_AND_DESCRIPTIONS, "r", encoding='utf-8') as f:
        return json.loads(f.read())

@st.cache_data
def get_all_participants():
    all_df = pd.read_csv(DATA, sep=';')
    df_with_clusters = predict_model(model, data=all_df)

    return df_with_clusters

@st.cache_resource
def find_similar_people(person_df, same_cluster_df, model): #zakladamy, ze model transformuje dane
    transformed_all = model.transform(same_cluster_df.drop(columns=["Cluster"]))
    transformed_user = model.transform(person_df)

    similarities = cosine_similarity(transformed_user, transformed_all)[0]

    same_cluster_df = same_cluster_df.copy()
    same_cluster_df["similarity"] = similarities

    return same_cluster_df.sort_values("similarity", ascending=False).head(5)

def make_histogram(df, column, title):
    fig = px.histogram(df, x=column)
    fig.update_layout(
        title=title,
        xaxis_title=column,
        yaxis_title="Liczba osób"
    )
    return fig

#profil grupy vs ogół
def compare_distributon(all_df, cluster_df, column):
    all_dist = all_df[column].value_counts(normalize=True)
    cluster_dist = cluster_df[column].value_counts(normalize=True)

    df = pd.DataFrame({
        "Ogół": all_dist,
        "Twoja grupa": cluster_dist    
    }).fillna(0)

    return df
